CREATE OR REPLACE VIEW tenant_revenue_ledger AS

SELECT v.tenant_id, COALESCE(SUM(li.price), 0.00)

FROM vehicles v

LEFT JOIN work_orders wo ON wo.vehicle_id = v.id

GROUP BY v.tenant_id;
